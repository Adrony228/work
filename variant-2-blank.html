а) Укажите ошибки.
б) Запишите исправленные строки.

n = 73462815
x = -1
y = -1
while n:
    a = n % 10
    m = n = n // 10
    while m:
        b = m % 10
        m = m // 10
        if a < b:
            if a * b >= x * y:
                x = b
                y = a
if x == -1:
    print(x)
else:
    print(x, y)
